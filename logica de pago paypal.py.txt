import datetime
import requests

# Configuración de PayPal Sandbox (Credenciales de prueba)
PAYPAL_CLIENT_ID = "TU_CLIENT_ID_SANDBOX"
PAYPAL_SECRET = "TU_SECRET_SANDBOX"
BASE_URL = "https://api-m.sandbox.paypal.com"  # Entorno de pruebas


class Factura:

    def __init__(self, id_factura, cliente, items):
        self.id_factura = id_factura
        self.cliente = cliente
        self.items = items  # Lista de diccionarios: [{'nombre': ..., 'precio': ...}]
        self.fecha = datetime.date.today().strftime("%Y-%m-%d")
        self.total = sum(item["precio"] for item in items)
        self.estado_pago = "PENDIENTE"

    def mostrar_factura(self):
        print(f"\n================ FACTURA #{self.id_factura} ================")
        print(f"Cliente: {self.cliente}")
        print(f"Fecha: {self.fecha}")
        print("-" * 44)
        for item in self.items:
            print(f" - {item['nombre']}: ${item['precio']:.2f}")
        print("-" * 44)
        print(f"TOTAL A PAGAR: ${self.total:.2f}")
        print(f"Estado de pago: {self.estado_pago}")
        print("================================================\n")


class PasarelaPayPal:

    def __init__(self, client_id, secret):
        self.client_id = client_id
        self.secret = secret

    def obtener_token_acceso(self):
        """Obtiene el Bearer Token para autenticarse con la API de PayPal Sandbox."""
        endpoint = f"{BASE_URL}/v1/oauth2/token"
        headers = {"Accept": "application/json", "Accept-Language": "en_US"}
        data = {"grant_type": "client_credentials"}

        response = requests.post(
            endpoint,
            headers=headers,
            data=data,
            auth=(self.client_id, self.secret),
        )

        if response.status_code == 200:
            return response.json()["access_token"]
        else:
            raise Exception(f"Error al obtener Token: {response.text}")

    def crear_orden_pago(self, factura):
        """Crea una orden de pago en PayPal vinculada a la factura."""
        token = self.obtener_token_acceso()
        endpoint = f"{BASE_URL}/v2/checkout/orders"

        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {token}",
        }

        payload = {
            "intent": "CAPTURE",
            "purchase_units": [
                {
                    "reference_id": str(factura.id_factura),
                    "description": f"Pago Factura #{factura.id_factura}",
                    "amount": {
                        "currency_code": "USD",
                        "value": f"{factura.total:.2f}",
                    },
                }
            ],
            "application_context": {
                "return_url": "https://ejemplo.com/retorno_exitoso",
                "cancel_url": "https://ejemplo.com/pago_cancelado",
            },
        }

        response = requests.post(endpoint, headers=headers, json=payload)

        if response.status_code == 201:
            datos_orden = response.json()
            order_id = datos_orden["id"]

            # Extraer el enlace de aprobación donde el cliente "iniciaría sesión" en PayPal
            approve_url = next(
                link["href"]
                for link in datos_orden["links"]
                if link["rel"] == "approve"
            )

            return order_id, approve_url
        else:
            raise Exception(f"Error al crear orden: {response.text}")
